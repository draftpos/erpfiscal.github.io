# Python Integration Guide

This guide shows how to communicate with HavanoZimra Cloud from Python using the popular `requests` library.

## 2. HavanoZimraClient Class

```python
import xml.etree.ElementTree as ET
from typing import Dict, List, Optional
import requests


class HavanoZimraClient:
    def __init__(self, base_url: str, api_key: str, api_secret: str, device_sn: str):
        self.base_url = base_url.rstrip("/")
        self.api_key = api_key
        self.api_secret = api_secret
        self.device_sn = device_sn
        self.session = requests.Session()

        # Seed guest cookies
        self.session.cookies.set("full_name", "Guest")
        self.session.cookies.set("sid", "Guest")
        self.session.cookies.set("system_user", "no")
        self.session.cookies.set("user_id", "Guest")
        self.csrf_token: Optional[str] = None

    def get_token(self) -> str:
        """Retrieves and caches the session CSRF token."""
        url = f"{self.base_url}/api/method/havanozimracloud.api.token"
        response = self.session.post(url)
        response.raise_for_status()

        data = response.json()
        token = data.get("token") or data.get("message")
        if not token:
            raise ValueError(f"Could not find token in response: {data}")

        self.csrf_token = token
        return token

    def _headers(self) -> Dict[str, str]:
        if not self.csrf_token:
            self.get_token()
        return {
            "Authorization": f"token {self.api_key}:{self.api_secret}",
            "X-Frappe-CSRF-Token": self.csrf_token,
            "Content-Type": "application/x-www-form-urlencoded",
        }

    def ping_zimra(self) -> dict:
        """Pings the ZIMRA server for the configured device."""
        url = f"{self.base_url}/api/method/havanozimracloud.api.pingzimra"
        data = {"device_sn": self.device_sn}
        response = self.session.post(url, headers=self._headers(), data=data)
        return response.json()

    def build_items_xml(self, items: List[dict]) -> str:
        """Utility to build compliant items_xml string."""
        root = ET.Element("ITEMS")
        for idx, item in enumerate(items, start=1):
            item_el = ET.SubElement(root, "ITEM")
            ET.SubElement(item_el, "HH").text = str(idx)
            ET.SubElement(item_el, "ITEMCODE").text = str(item["code"])
            ET.SubElement(item_el, "ITEMNAME").text = str(item["name"])
            ET.SubElement(item_el, "ITEMNAME2").text = str(item.get("name2", item["name"]))
            ET.SubElement(item_el, "QTY").text = str(item["qty"])
            ET.SubElement(item_el, "PRICE").text = f"{float(item['price']):.2f}"
            ET.SubElement(item_el, "TOTAL").text = f"{float(item['total']):.2f}"
            ET.SubElement(item_el, "VAT").text = f"{float(item['vat']):.2f}"
            ET.SubElement(item_el, "VATR").text = f"{float(item['vatr']):.2f}"
            ET.SubElement(item_el, "VNAME").text = str(item["vname"])
        return ET.tostring(root, encoding="utf-8").decode("utf-8")

    def send_invoice(self, payload: dict) -> dict:
        """Sends a sales invoice or credit note to ZIMRA."""
        url = f"{self.base_url}/api/method/havanozimracloud.api.sendinvoice"
        if "device_sn" not in payload:
            payload["device_sn"] = self.device_sn

        response = self.session.post(url, headers=self._headers(), data=payload)
        return response.json()
```

## 3. Example Execution

```python
if __name__ == "__main__":
    client = HavanoZimraClient(
        base_url="https://your-site-name",
        api_key="YOUR_API_KEY",
        api_secret="YOUR_API_SECRET",
        device_sn="1234567890",
    )

    # 1. Ping device
    ping = client.ping_zimra()
    print("Ping:", ping)

    # 2. Build XML items
    items = [
        {
            "code": "99001868",
            "name": "Milo 500g",
            "qty": 1,
            "price": 6.10,
            "total": 6.10,
            "vat": 0.80,
            "vatr": 0.15,
            "vname": "VAT",
        }
    ]
    xml_string = client.build_items_xml(items)

    # 3. Send invoice
    invoice_payload = {
        "add_customer": 0,
        "invoice_flag": 0,
        "currency": "USD",
        "invoice_number": "INV-PY-001",
        "customer_name": "Tendai Moyo",
        "trade_name": "Moyo Enterprises",
        "customer_vat_number": "12345678",
        "customer_address": "88 Nelson Mandela Ave",
        "customer_telephone_number": "+263773111222",
        "customer_tin": "76543210",
        "customer_province": "Harare",
        "customer_street": "Nelson Mandela",
        "customer_houseNo": "88",
        "customer_city": "Harare",
        "customer_email": "tendai@moyo.co.zw",
        "invoice_comment": "Wholesale purchase",
        "original_invoice_no": "",
        "global_invoice_no": "",
        "items_xml": xml_string,
    }

    result = client.send_invoice(invoice_payload)
    print("Submission Result:", result)
```
