# C# (.NET) Integration Guide

This guide demonstrates how to integrate modern .NET applications (C# 10 / 12 on .NET 6, 8, or 9) with HavanoZimra Cloud.

---

## 1. HavanoZimraClient Service Implementation

```csharp
using System.Net;
using System.Text.Json;
using System.Text.Json.Nodes;

namespace HavanoZimra.Integration;

public class HavanoZimraClient
{
    private readonly HttpClient _httpClient;
    private readonly string _baseUrl;
    private readonly string _apiKey;
    private readonly string _apiSecret;
    private readonly string _deviceSn;
    private string? _cachedToken;

    public HavanoZimraClient(string baseUrl, string apiKey, string apiSecret, string deviceSn)
    {
        _baseUrl = baseUrl.TrimEnd('/');
        _apiKey = apiKey;
        _apiSecret = apiSecret;
        _deviceSn = deviceSn;

        var handler = new HttpClientHandler
        {
            UseCookies = true,
            CookieContainer = new CookieContainer()
        };

        var baseUri = new Uri(_baseUrl);
        handler.CookieContainer.Add(baseUri, new Cookie("full_name", "Guest"));
        handler.CookieContainer.Add(baseUri, new Cookie("sid", "Guest"));
        handler.CookieContainer.Add(baseUri, new Cookie("system_user", "no"));
        handler.CookieContainer.Add(baseUri, new Cookie("user_id", "Guest"));

        _httpClient = new HttpClient(handler);
    }

    /// <summary>
    /// Retrieves or refreshes the CSRF token.
    /// </summary>
    public async Task<string> GetTokenAsync(CancellationToken cancellationToken = default)
    {
        var url = $"{_baseUrl}/api/method/havanozimracloud.api.token";
        var response = await _httpClient.GetAsync(url, cancellationToken);
        response.EnsureSuccessStatusCode();

        var content = await response.Content.ReadAsStringAsync(cancellationToken);
        var json = JsonNode.Parse(content);

        _cachedToken = json?["token"]?.ToString() ?? json?["message"]?.ToString();
        if (string.IsNullOrEmpty(_cachedToken))
        {
            throw new InvalidOperationException($"Unable to parse CSRF token from response: {content}");
        }

        return _cachedToken;
    }

    /// <summary>
    /// Pings the ZIMRA server for the configured device.
    /// </summary>
    public async Task<string> PingZimraAsync(CancellationToken cancellationToken = default)
    {
        var token = _cachedToken ?? await GetTokenAsync(cancellationToken);
        var url = $"{_baseUrl}/api/method/havanozimracloud.api.pingzimra";

        using var request = new HttpRequestMessage(HttpMethod.Post, url);
        request.Headers.Add("X-Frappe-CSRF-Token", token);
        request.Headers.Add("Authorization", $"token {_apiKey}:{_apiSecret}");

        var payload = new Dictionary<string, string>
        {
            ["device_sn"] = _deviceSn
        };
        request.Content = new FormUrlEncodedContent(payload);

        var response = await _httpClient.SendAsync(request, cancellationToken);
        return await response.Content.ReadAsStringAsync(cancellationToken);
    }

    /// <summary>
    /// Sends a sales invoice or credit note to ZIMRA.
    /// </summary>
    public async Task<string> SendInvoiceAsync(
        Dictionary<string, string> parameters,
        CancellationToken cancellationToken = default)
    {
        var token = _cachedToken ?? await GetTokenAsync(cancellationToken);
        var url = $"{_baseUrl}/api/method/havanozimracloud.api.sendinvoice";

        using var request = new HttpRequestMessage(HttpMethod.Post, url);
        request.Headers.Add("X-Frappe-CSRF-Token", token);
        request.Headers.Add("Authorization", $"token {_apiKey}:{_apiSecret}");

        // Automatically inject device serial number if omitted
        if (!parameters.ContainsKey("device_sn"))
        {
            parameters["device_sn"] = _deviceSn;
        }

        request.Content = new FormUrlEncodedContent(parameters);

        var response = await _httpClient.SendAsync(request, cancellationToken);
        return await response.Content.ReadAsStringAsync(cancellationToken);
    }
}
```

## 2. Usage in ASP.NET Core or Console

```csharp
var client = new HavanoZimraClient(
    baseUrl: "https://your-site-name",
    apiKey: "YOUR_API_KEY",
    apiSecret: "YOUR_API_SECRET",
    deviceSn: "1234567890"
);

// 1. Verify connection
var pingResult = await client.PingZimraAsync();
Console.WriteLine($"Ping Response: {pingResult}");

// 2. Build invoice parameters
var invoiceData = new Dictionary<string, string>
{
    ["add_customer"] = "0",
    ["invoice_flag"] = "0",
    ["currency"] = "USD",
    ["invoice_number"] = $"INV-{DateTime.UtcNow:yyyyMMddHHmmss}",
    ["customer_name"] = "Acme Supplies Ltd",
    ["trade_name"] = "Acme Supplies",
    ["customer_vat_number"] = "10023456",
    ["customer_address"] = "50 Robert Mugabe Way",
    ["customer_telephone_number"] = "+263772000111",
    ["customer_tin"] = "20005432",
    ["customer_province"] = "Harare",
    ["customer_street"] = "Robert Mugabe Way",
    ["customer_houseNo"] = "50",
    ["customer_city"] = "Harare",
    ["customer_email"] = "accounts@acme.co.zw",
    ["invoice_comment"] = "Office equipment sales",
    ["original_invoice_no"] = "",
    ["global_invoice_no"] = "",
    ["items_xml"] = @"<ITEMS>
        <ITEM>
            <HH>1</HH>
            <ITEMCODE>PRN01</ITEMCODE>
            <ITEMNAME>Laser Printer</ITEMNAME>
            <ITEMNAME2>Laser Printer</ITEMNAME2>
            <QTY>1</QTY>
            <PRICE>150.00</PRICE>
            <TOTAL>150.00</TOTAL>
            <VAT>19.57</VAT>
            <VATR>0.15</VATR>
            <VNAME>VAT</VNAME>
        </ITEM>
    </ITEMS>"
};

// 3. Send to ZIMRA
var invoiceResult = await client.SendInvoiceAsync(invoiceData);
Console.WriteLine($"Invoice Result: {invoiceResult}");
```
