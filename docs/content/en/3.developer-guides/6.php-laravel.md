# PHP & Laravel Integration

This guide provides a clean object-oriented PHP 8+ / Laravel client to communicate with HavanoZimra Cloud using Guzzle or native cURL.

---

## 1. HavanoZimraService Implementation

```php
<?php

namespace App\Services;

use GuzzleHttp\Client;
use GuzzleHttp\Cookie\CookieJar;
use Exception;

class HavanoZimraService
{
    protected string $baseUrl;
    protected string $apiKey;
    protected string $apiSecret;
    protected string $deviceSn;
    protected Client $httpClient;
    protected CookieJar $cookieJar;
    protected ?string $csrfToken = null;

    public function __construct(string $baseUrl, string $apiKey, string $apiSecret, string $deviceSn)
    {
        $this->baseUrl = rtrim($baseUrl, '/');
        $this->apiKey = $apiKey;
        $this->apiSecret = $apiSecret;
        $this->deviceSn = $deviceSn;
        $this->cookieJar = new CookieJar();

        $this->httpClient = new Client([
            'base_uri' => $this->baseUrl,
            'cookies' => $this->cookieJar,
            'timeout' => 30.0,
            'http_errors' => false,
        ]);
    }

    /**
     * Retrieve and cache CSRF token
     */
    public function getCsrfToken(): string
    {
        $response = $this->httpClient->post('/api/method/havanozimracloud.api.token');
        $body = json_decode((string)$response->getBody(), true);

        $token = $body['token'] ?? $body['message'] ?? null;
        if (!$token) {
            throw new Exception("Unable to retrieve CSRF token: " . (string)$response->getBody());
        }

        $this->csrfToken = $token;
        return $token;
    }

    /**
     * Ping ZIMRA to check device connectivity
     */
    public function ping(): array
    {
        $token = $this->csrfToken ?: $this->getCsrfToken();

        $response = $this->httpClient->post('/api/method/havanozimracloud.api.pingzimra', [
            'headers' => [
                'Authorization' => "token {$this->apiKey}:{$this->apiSecret}",
                'X-Frappe-CSRF-Token' => $token,
                'Content-Type' => 'application/x-www-form-urlencoded',
            ],
            'form_params' => [
                'device_sn' => $this->deviceSn,
            ],
        ]);

        return json_decode((string)$response->getBody(), true);
    }

    /**
     * Build the structured items_xml string
     */
    public function buildItemsXml(array $items): string
    {
        $xml = new \SimpleXMLElement('<ITEMS/>');

        foreach ($items as $index => $item) {
            $row = $xml->addChild('ITEM');
            $row->addChild('HH', (string)($index + 1));
            $row->addChild('ITEMCODE', htmlspecialchars($item['code']));
            $row->addChild('ITEMNAME', htmlspecialchars($item['name']));
            $row->addChild('ITEMNAME2', htmlspecialchars($item['name2'] ?? $item['name']));
            $row->addChild('QTY', number_format((float)$item['qty'], 2, '.', ''));
            $row->addChild('PRICE', number_format((float)$item['price'], 2, '.', ''));
            $row->addChild('TOTAL', number_format((float)$item['total'], 2, '.', ''));
            $row->addChild('VAT', number_format((float)$item['vat'], 2, '.', ''));
            $row->addChild('VATR', number_format((float)$item['vatr'], 2, '.', ''));
            $row->addChild('VNAME', htmlspecialchars($item['vname']));
        }

        return $xml->asXML();
    }

    /**
     * Submit invoice or credit note
     */
    public function sendInvoice(array $payload): array
    {
        $token = $this->csrfToken ?: $this->getCsrfToken();

        if (!isset($payload['device_sn'])) {
            $payload['device_sn'] = $this->deviceSn;
        }

        $response = $this->httpClient->post('/api/method/havanozimracloud.api.sendinvoice', [
            'headers' => [
                'Authorization' => "token {$this->apiKey}:{$this->apiSecret}",
                'X-Frappe-CSRF-Token' => $token,
                'Content-Type' => 'application/x-www-form-urlencoded',
            ],
            'form_params' => $payload,
        ]);

        return json_decode((string)$response->getBody(), true);
    }
}
```

## 2. Usage in a Controller or Command

```php
$zimra = new HavanoZimraService(
    baseUrl: 'https://your-site-name',
    apiKey: config('services.zimra.key'),
    apiSecret: config('services.zimra.secret'),
    deviceSn: '1234567890'
);

// Ping test
$ping = $zimra->ping();

// Build items
$itemsXml = $zimra->buildItemsXml([
    [
        'code' => 'SKU-001',
        'name' => 'Solar Battery 100Ah',
        'qty' => 1,
        'price' => 220.00,
        'total' => 220.00,
        'vat' => 28.70,
        'vatr' => 0.15,
        'vname' => 'VAT',
    ],
]);

// Send invoice
$result = $zimra->sendInvoice([
    'add_customer' => 0,
    'invoice_flag' => 0,
    'currency' => 'USD',
    'invoice_number' => 'INV-' . time(),
    'customer_name' => 'Tafadzwa Chitepo',
    'trade_name' => 'Chitepo Tech',
    'customer_vat_number' => '12345678',
    'customer_address' => '45 Kwame Nkrumah Ave',
    'customer_telephone_number' => '+263774888999',
    'customer_tin' => '88990011',
    'customer_province' => 'Harare',
    'customer_street' => 'Kwame Nkrumah',
    'customer_houseNo' => '45',
    'customer_city' => 'Harare',
    'customer_email' => 'tafadzwa@chitepotech.co.zw',
    'invoice_comment' => 'Hardware invoice',
    'original_invoice_no' => '',
    'global_invoice_no' => '',
    'items_xml' => $itemsXml,
]);

if (($result['ResponseCode'] ?? null) === '0') {
    // Save $result['GlobalInvoiceNo'] and print receipt
}
```
