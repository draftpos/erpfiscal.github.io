# VB.NET Integration Guide

This guide provides tested production code for integrating VB.NET applications (WinForms, WPF, ASP.NET, Windows Services) with HavanoZimra Cloud.

## 1. Get CSRF Token Function

The `GetToken` function initializes security protocols (TLS 1.2 / 1.3), sets default session cookies, sends a GET request to `/api/method/havanozimracloud.api.token`, and parses the returned token using `Newtonsoft.Json`.

```vb
Imports System.IO
Imports System.Net
Imports System.Net.Http
Imports System.Threading.Tasks
Imports Newtonsoft.Json

Public Module HavanoZimraClient

    Public hcloud_baseurl As String = "https://your-site-name"
    Public hcloud_key As String = "YOUR_API_KEY"
    Public hcloud_secret As String = "YOUR_API_SECRET"
    Public hcloud_devicesn As String = "1234567890"

    Public Async Function GetToken() As Task(Of String)
        Dim apiUrl As String = hcloud_baseurl & "/api/method/havanozimracloud.api.token"
        Debug.WriteLine(apiUrl)
        Dim result As String = ""
        Try
            ' Ensure TLS 1.2 or TLS 1.3 is enforced
            ServicePointManager.SecurityProtocol = SecurityProtocolType.Tls12 Or SecurityProtocolType.Tls13
            ServicePointManager.ServerCertificateValidationCallback = Function(sender, certificate, chain, sslPolicyErrors) True

            Dim cookieContainer As New CookieContainer()
            cookieContainer.Add(New Uri(apiUrl), New Cookie("full_name", "Guest"))
            cookieContainer.Add(New Uri(apiUrl), New Cookie("sid", "Guest"))
            cookieContainer.Add(New Uri(apiUrl), New Cookie("system_user", "no"))
            cookieContainer.Add(New Uri(apiUrl), New Cookie("user_id", "Guest"))
            cookieContainer.Add(New Uri(apiUrl), New Cookie("user_image", ""))

            Dim handler As New HttpClientHandler() With {
                .UseCookies = True,
                .CookieContainer = cookieContainer
            }

            Using httpClient As New HttpClient(handler)
                Dim httpRequestMessage As New HttpRequestMessage(HttpMethod.Get, apiUrl)
                Dim response As HttpResponseMessage = Await httpClient.SendAsync(httpRequestMessage)

                If response.IsSuccessStatusCode Then
                    result = Await response.Content.ReadAsStringAsync()
                    Debug.WriteLine(result)
                    Dim data = JsonConvert.DeserializeObject(Of Dictionary(Of String, String))(result)

                    If data.ContainsKey("token") Then
                        Return data("token")
                    ElseIf data.ContainsKey("message") Then
                        Return data("message")
                    Else
                        Return "No token or message key in response: " & result
                    End If
                Else
                    result = Await response.Content.ReadAsStringAsync()
                    Return result
                End If
            End Using

        Catch ex As Exception
            Debug.WriteLine("Error retrieving token: " & ex.Message)
            Return "Failed"
        End Try

        Return result
    End Function

End Module
```

## 2. Send Invoice Function

The `SendInvoice2Cloud` function transmits fiscal invoices and credit notes to `/api/method/havanozimracloud.api.sendinvoice` using `HttpWebRequest` with `application/x-www-form-urlencoded` encoding.

```vb
Public Async Function SendInvoice2Cloud(
    ByVal AddCustomer As String,
    ByVal InvoiceFlag As String,
    ByVal Currency As String,
    ByVal InvoiceNumber As String,
    ByVal CustomerName As String,
    ByVal TradeName As String,
    ByVal CustomerVATNumber As String,
    ByVal CustomerAddress As String,
    ByVal CustomerTelephoneNumber As String,
    ByVal CustomerTIN As String,
    ByVal CustomerProvince As String,
    ByVal CustomerStreet As String,
    ByVal CustomerHouseNo As String,
    ByVal CustomerCity As String,
    ByVal CustomerEmail As String,
    ByVal InvoiceComment As String,
    ByVal OriginalInvoiceNo As String,
    ByVal GlobalInvoiceNo As String,
    ByVal ItemsXML As String,
    ByVal ftoken As String
) As Task(Of String)

    Debug.WriteLine("Sending Request to HavanoZimra")
    Dim url As String = hcloud_baseurl + "/api/method/havanozimracloud.api.sendinvoice"
    Dim result As String = ""

    Dim postData As String =
        $"device_sn={Uri.EscapeDataString(hcloud_devicesn)}&" &
        $"add_customer={Uri.EscapeDataString(AddCustomer)}&" &
        $"invoice_flag={Uri.EscapeDataString(InvoiceFlag)}&" &
        $"currency={Uri.EscapeDataString(Currency)}&" &
        $"invoice_number={Uri.EscapeDataString(InvoiceNumber)}&" &
        $"customer_name={Uri.EscapeDataString(CustomerName)}&" &
        $"trade_name={Uri.EscapeDataString(TradeName)}&" &
        $"customer_vat_number={Uri.EscapeDataString(CustomerVATNumber)}&" &
        $"customer_address={Uri.EscapeDataString(CustomerAddress)}&" &
        $"customer_telephone_number={Uri.EscapeDataString(CustomerTelephoneNumber)}&" &
        $"customer_tin={Uri.EscapeDataString(CustomerTIN)}&" &
        $"customer_province={Uri.EscapeDataString(CustomerProvince)}&" &
        $"customer_street={Uri.EscapeDataString(CustomerStreet)}&" &
        $"customer_house_no={Uri.EscapeDataString(CustomerHouseNo)}&" &
        $"customer_city={Uri.EscapeDataString(CustomerCity)}&" &
        $"customer_email={Uri.EscapeDataString(CustomerEmail)}&" &
        $"invoice_comment={Uri.EscapeDataString(InvoiceComment)}&" &
        $"original_invoice_no={Uri.EscapeDataString(OriginalInvoiceNo)}&" &
        $"global_invoice_no={Uri.EscapeDataString(GlobalInvoiceNo)}&" &
        $"items_xml={Uri.EscapeDataString(ItemsXML)}"

    Dim request As HttpWebRequest = CType(WebRequest.Create(url), HttpWebRequest)
    request.Method = "POST"

    request.Headers("X-Frappe-CSRF-Token") = $"{ftoken}"
    request.Headers("Authorization") = $"token {hcloud_key}:{hcloud_secret}"
    request.ContentType = "application/x-www-form-urlencoded"

    Dim dataBytes As Byte() = System.Text.Encoding.UTF8.GetBytes(postData)
    request.ContentLength = dataBytes.Length

    Using requestStream As Stream = request.GetRequestStream()
        requestStream.Write(dataBytes, 0, dataBytes.Length)
    End Using

    Using response As HttpWebResponse = CType(request.GetResponse(), HttpWebResponse)
        Using reader As New StreamReader(response.GetResponseStream())
            result = reader.ReadToEnd()
        End Using
    End Using

    Return result
End Function
```

## 3. End-to-End Usage Example

```vb
Public Async Sub ProcessFiscalSale()
    ' 1. Fetch CSRF token
    Dim token As String = Await GetToken()
    If token = "Failed" OrElse String.IsNullOrEmpty(token) Then
        MessageBox.Show("Failed to obtain CSRF token from HavanoZimra Cloud")
        Return
    End If

    ' 2. Construct items XML
    Dim itemsXml As String =
        "<ITEMS>" &
            "<ITEM>" &
                "<HH>1</HH>" &
                "<ITEMCODE>99001868</ITEMCODE>" &
                "<ITEMNAME>Milo 500g</ITEMNAME>" &
                "<ITEMNAME2>Milo</ITEMNAME2>" &
                "<QTY>1</QTY>" &
                "<PRICE>6.10</PRICE>" &
                "<TOTAL>6.10</TOTAL>" &
                "<VAT>0.80</VAT>" &
                "<VATR>0.15</VATR>" &
                "<VNAME>VAT</VNAME>" &
            "</ITEM>" &
        "</ITEMS>"

    ' 3. Submit invoice
    Dim responseJson As String = Await SendInvoice2Cloud(
        AddCustomer:="0",
        InvoiceFlag:="0",
        Currency:="USD",
        InvoiceNumber:="INV-" & DateTime.Now.Ticks.ToString().Substring(10),
        CustomerName:="John Doe",
        TradeName:="JD Traders",
        CustomerVATNumber:="12345678",
        CustomerAddress:="123 Market Street",
        CustomerTelephoneNumber:="+263771234567",
        CustomerTIN:="87654321",
        CustomerProvince:="Harare",
        CustomerStreet:="Market Street",
        CustomerHouseNo:="12",
        CustomerCity:="Harare",
        CustomerEmail:="john@example.com",
        InvoiceComment:="Regular sale",
        OriginalInvoiceNo:="",
        GlobalInvoiceNo:="",
        ItemsXML:=itemsXml,
        ftoken:=token
    )

    Debug.WriteLine("Server Response: " & responseJson)
End Sub
```
