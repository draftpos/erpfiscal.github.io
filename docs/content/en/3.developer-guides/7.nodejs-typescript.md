# Node.js & TypeScript Integration

This guide provides a TypeScript client using native `fetch` or Axios for integrating Node.js backends and modern web APIs with HavanoZimra Cloud.

---

## 1. HavanoZimraClient in TypeScript

```typescript
export interface InvoiceItem {
  code: string;
  name: string;
  name2?: string;
  qty: number;
  price: number;
  total: number;
  vat: number;
  vatr: number;
  vname: 'VAT' | 'ZERO RATED' | 'EXEMPT';
}

export interface SendInvoiceOptions {
  add_customer: 0 | 1;
  invoice_flag: 0 | 1; // 0 = Invoice, 1 = Credit Note
  currency: string;
  invoice_number: string;
  customer_name: string;
  trade_name: string;
  customer_vat_number: string;
  customer_address: string;
  customer_telephone_number: string;
  customer_tin: string;
  customer_province: string;
  customer_street: string;
  customer_houseNo: string;
  customer_city: string;
  customer_email: string;
  invoice_comment?: string;
  original_invoice_no?: string;
  global_invoice_no?: string | number;
  items: InvoiceItem[];
}

export class HavanoZimraClient {
  private baseUrl: string;
  private apiKey: string;
  private apiSecret: string;
  private deviceSn: string;
  private cachedToken: string | null = null;

  constructor(config: { baseUrl: string; apiKey: string; apiSecret: string; deviceSn: string }) {
    this.baseUrl = config.baseUrl.replace(/\/+$/, '');
    this.apiKey = config.apiKey;
    this.apiSecret = config.apiSecret;
    this.deviceSn = config.deviceSn;
  }

  /**
   * Fetch and cache session CSRF token
   */
  async getToken(): Promise<string> {
    const url = `${this.baseUrl}/api/method/havanozimracloud.api.token`;
    const res = await fetch(url, {
      method: 'POST',
      headers: {
        Cookie: 'full_name=Guest; sid=Guest; system_user=no; user_id=Guest',
      },
    });

    if (!res.ok) {
      throw new Error(`Failed to fetch CSRF token: ${res.statusText}`);
    }

    const data = await res.json();
    const token = data.token || data.message;
    if (!token) {
      throw new Error(`Invalid token response payload: ${JSON.stringify(data)}`);
    }

    this.cachedToken = token;
    return token;
  }

  /**
   * Ping ZIMRA to verify connectivity
   */
  async pingZimra(): Promise<any> {
    const token = this.cachedToken || (await this.getToken());
    const url = `${this.baseUrl}/api/method/havanozimracloud.api.pingzimra`;

    const body = new URLSearchParams({
      device_sn: this.deviceSn,
    });

    const res = await fetch(url, {
      method: 'POST',
      headers: {
        'Authorization': `token ${this.apiKey}:${this.apiSecret}`,
        'X-Frappe-CSRF-Token': token,
        'Content-Type': 'application/x-www-form-urlencoded',
      },
      body: body.toString(),
    });

    return res.json();
  }

  /**
   * Build XML string from item objects
   */
  buildItemsXml(items: InvoiceItem[]): string {
    const lines = items
      .map(
        (it, idx) => `  <ITEM>
    <HH>${idx + 1}</HH>
    <ITEMCODE>${this.escapeXml(it.code)}</ITEMCODE>
    <ITEMNAME>${this.escapeXml(it.name)}</ITEMNAME>
    <ITEMNAME2>${this.escapeXml(it.name2 || it.name)}</ITEMNAME2>
    <QTY>${it.qty.toFixed(2)}</QTY>
    <PRICE>${it.price.toFixed(2)}</PRICE>
    <TOTAL>${it.total.toFixed(2)}</TOTAL>
    <VAT>${it.vat.toFixed(2)}</VAT>
    <VATR>${it.vatr.toFixed(2)}</VATR>
    <VNAME>${this.escapeXml(it.vname)}</VNAME>
  </ITEM>`
      )
      .join('\n');

    return `<ITEMS>\n${lines}\n</ITEMS>`;
  }

  private escapeXml(unsafe: string): string {
    return unsafe.replace(/[<>&'"]/g, (c) => {
      switch (c) {
        case '<': return '&lt;';
        case '>': return '&gt;';
        case '&': return '&amp;';
        case '\'': return '&apos;';
        case '"': return '&quot;';
        default: return c;
      }
    });
  }

  /**
   * Submit fiscal invoice or credit note
   */
  async sendInvoice(options: SendInvoiceOptions): Promise<any> {
    const token = this.cachedToken || (await this.getToken());
    const url = `${this.baseUrl}/api/method/havanozimracloud.api.sendinvoice`;

    const params = new URLSearchParams({
      device_sn: this.deviceSn,
      add_customer: String(options.add_customer),
      invoice_flag: String(options.invoice_flag),
      currency: options.currency,
      invoice_number: options.invoice_number,
      customer_name: options.customer_name,
      trade_name: options.trade_name,
      customer_vat_number: options.customer_vat_number,
      customer_address: options.customer_address,
      customer_telephone_number: options.customer_telephone_number,
      customer_tin: options.customer_tin,
      customer_province: options.customer_province,
      customer_street: options.customer_street,
      customer_houseNo: options.customer_houseNo,
      customer_city: options.customer_city,
      customer_email: options.customer_email,
      invoice_comment: options.invoice_comment || '',
      original_invoice_no: options.original_invoice_no || '',
      global_invoice_no: options.global_invoice_no ? String(options.global_invoice_no) : '',
      items_xml: this.buildItemsXml(options.items),
    });

    const res = await fetch(url, {
      method: 'POST',
      headers: {
        'Authorization': `token ${this.apiKey}:${this.apiSecret}`,
        'X-Frappe-CSRF-Token': token,
        'Content-Type': 'application/x-www-form-urlencoded',
      },
      body: params.toString(),
    });

    return res.json();
  }
}
```

## 2. Usage Example

```typescript
const zimra = new HavanoZimraClient({
  baseUrl: 'https://your-site-name',
  apiKey: process.env.HAVANO_API_KEY!,
  apiSecret: process.env.HAVANO_API_SECRET!,
  deviceSn: '1234567890',
});

async function main() {
  const ping = await zimra.pingZimra();
  console.log('ZIMRA Ping:', ping);

  const response = await zimra.sendInvoice({
    add_customer: 0,
    invoice_flag: 0,
    currency: 'USD',
    invoice_number: `INV-${Date.now()}`,
    customer_name: 'Farai Ndebele',
    trade_name: 'Ndebele Retail',
    customer_vat_number: '12345678',
    customer_address: '10 Jason Moyo Ave',
    customer_telephone_number: '+263777000111',
    customer_tin: '55443322',
    customer_province: 'Bulawayo',
    customer_street: 'Jason Moyo',
    customer_houseNo: '10',
    customer_city: 'Bulawayo',
    customer_email: 'farai@retail.co.zw',
    items: [
      {
        code: '99001868',
        name: 'Milo 500g',
        qty: 2,
        price: 6.10,
        total: 12.20,
        vat: 1.59,
        vatr: 0.15,
        vname: 'VAT',
      },
    ],
  });

  console.log('Fiscal Output:', response);
}

main().catch(console.error);
```
